// Todo 아이템을 렌더링하는 함수를 분리 function renderTodoItem(todo) { return `

Name: ${todo.name}

Description: ${todo.description}

Complete:

Completed At: ${datetimeToString(todo.completed_at)}

Experience Points: ${todo.exp}

`; } // Todo 리스트를 불러오고 렌더링하는 함수 function loadAndRenderTodos() { return axiosInstance.get("/api/todo/list/") .then(function(response) { const todoHTML = response.data.map(renderTodoItem).join(''); $(".listcontainer").find('.todo-item').remove(); $(".listcontainer").append(todoHTML); }) .catch(function(error) { console.log(error); }); } // 페이지 로드시 $(document).ready(loadAndRenderTodos); function detailView(id) { window.location.href = `/todo/${id}/`; } function toComplete(id, event) { event.stopPropagation(); // 현재 완료한 todo 아이템의 위치를 찾습니다 const completedItem = $(event.target).closest('.todo-item'); const itemOffset = completedItem.offset().top - $(window).scrollTop(); axiosInstance.patch(`/api/todo/update/${id}/`, { complete: true, }) .then(function(response) { // 데이터를 새로 불러온 후 스크롤 위치를 조정합니다 loadAndRenderTodos().then(() => { const newItem = $(`.todo-item:contains(${completedItem.find('p:first').text()})`); if (newItem.length) { const newScrollTop = newItem.offset().top - itemOffset; $(window).scrollTop(newScrollTop); } }); }) .catch(function(error) { console.log(error); }); }
